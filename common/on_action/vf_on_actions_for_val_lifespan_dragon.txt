on_dragon_birthday = {
	on_actions = {
		vf_apply_valyrian_dragon_modifier
	}
}

agot_yearly_dragon_pulse = {
	on_actions = {
		vf_valyrian_modifier_count
	}
}

vf_valyrian_modifier_count = {
	trigger = {
		AND = {
			vf_dragon_or_its_rider_lives_in_valyria = no
			vf_has_valyrian_life_span = yes
		}
	}
	effect = {
		# BUG-16 FIX: Restructured counter logic to avoid race condition.
		# The second 'if' was independent, causing the modifier to be removed
		# in the same tick it was incremented to 6. Now using else_if to ensure
		# proper sequential evaluation.
		if = {
			limit = {
				NOT = { exists = var:vf_count_to_five }
			}
			set_variable = {
				name = vf_count_to_five
				value = 1
			}
		}
		else_if = {
			limit = {
				exists = var:vf_count_to_five
				var:vf_count_to_five < 6
			}
			change_variable = {
				name = vf_count_to_five
				add = 1
			}
		}
		else_if = {
			# BUG-16 FIX: Only remove after the counter has been >= 6 for a full tick
			limit = {
				var:vf_count_to_five >= 6
			}
			remove_character_modifier = vf_lives_in_valyria_modifier
			remove_variable = vf_count_to_five
		}
	}
}

vf_apply_valyrian_dragon_modifier = {
	trigger = {
		vf_dragon_or_its_rider_lives_in_valyria = yes
	}
	effect = {
		# BUG-06 FIX: Only add the modifier if it is not already present.
		# Without this check, every dragon birthday would stack another permanent
		# instance of vf_lives_in_valyria_modifier, causing massive unintended bonuses.
		if = {
			limit = {
				NOT = { has_character_modifier = vf_lives_in_valyria_modifier }
			}
			add_character_modifier = {
				modifier = vf_lives_in_valyria_modifier
			}
		}
	}
}
